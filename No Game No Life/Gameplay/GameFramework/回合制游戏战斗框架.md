### 第一层：Entity / Tag 模型（Fireplace 核心思路）

所有游戏对象都是 Entity 的子类，每个 Entity 通过一个 tags 字典追踪其公开属性，存储 GameTag 键与 bool/int/str/Entity 值的键值对。游戏由 Game、Player 和 Card 对象组成，每张卡有一个 Zone（可以是 PLAY、DECK、HAND、GRAVEYARD、REMOVEDFROMGAME、SETASIDE 或 SECRET）。

这个设计的精髓在于：**没有继承爆炸，没有 `MinionWithTauntAndDivineShield` 这种类**。所有属性都是 tag，逻辑层读写 tag，而非依赖类型系统。

```csharp
// C# 生产级 Entity-Tag 模型
public class GameEntity
{
    private Dictionary<GameTag, int> _tags = new();

    public int GetTag(GameTag tag) => _tags.TryGetValue(tag, out var v) ? v : 0;
    public void SetTag(GameTag tag, int value)
    {
        _tags[tag] = value;
        EventBus.Emit(new TagChangedEvent(this, tag, value)); // 同步给客户端
    }
    public bool HasTag(GameTag tag) => GetTag(tag) > 0;
}

public enum GameTag
{
    ATK, HEALTH, MAX_HEALTH, COST, ZONE,
    TAUNT, DIVINE_SHIELD, CHARGE, WINDFURY,
    DEATHRATTLE, BATTLECRY, COMBO,
    TURN, STEP, CURRENT_PLAYER,
    ENTITY_ID, CONTROLLER, CARDTYPE
}
```

### 第二层：Selector DSL + Atomic Actions（卡牌效果描述语言）

这是最关键的部分。Fireplace 的核心突破是将"对友方随从造成2点伤害"和"对敌方随从造成2点伤害"统一为同一个 `Hit` action 加不同的 `Selector`，而非两套不同的辅助函数。

```csharp
// ===== Selector：描述"谁" =====
[Flags]
public enum TargetSelector
{
    FRIENDLY  = 1 << 0,
    ENEMY     = 1 << 1,
    MINION    = 1 << 2,
    HERO      = 1 << 3,
    ALL       = 1 << 4,
    RANDOM    = 1 << 5,
    DAMAGED   = 1 << 6,
    LOWEST_HP = 1 << 7,
    ADJACENT  = 1 << 8,
}

// ===== Atomic Actions：描述"做什么" =====
// 这是 Composite Pattern 的叶节点
public abstract record GameAction;

public record Hit(TargetSelector Target, int Amount) : GameAction;
public record Heal(TargetSelector Target, int Amount) : GameAction;
public record Buff(TargetSelector Target, int AttackBonus, int HealthBonus) : GameAction;
public record DrawCards(int Count) : GameAction;
public record SummonMinion(string CardId, TargetSelector Position) : GameAction;
public record Silence(TargetSelector Target) : GameAction;
public record Destroy(TargetSelector Target) : GameAction;
public record AddToHand(string CardId) : GameAction;
public record TransformInto(TargetSelector Target, string CardId) : GameAction;

// ===== 卡牌定义：数据驱动，非硬编码 =====
// 参考 Fireplace 的 DSL 风格
// class CS2_032 (Flamestrike):
//   action = Hit(ENEMY | MINION, 4)
//
// class EX1_046 (Dark Iron Dwarf):
//   action = Buff(TARGET | FRIENDLY | MINION, attackBonus: 2, healthBonus: 0)
//
// class EX1_112 (Gelbin Mekkatorque):
//   deathrattle = SummonMinion("EX1_112t", FRIENDLY | BOARD)

public class CardDefinition
{
    public string CardId { get; init; }
    public int Cost { get; init; }
    public CardType Type { get; init; }

    // 核心触发钩子
    public IReadOnlyList<GameAction> BattlecryActions { get; init; }
    public IReadOnlyList<GameAction> DeathrattleActions { get; init; }
    public IReadOnlyList<GameAction> ComboActions { get; init; }   // 本回合已出过牌
    public IReadOnlyList<GameAction> InspireActions { get; init; } // 使用英雄技能后

    // 事件监听（持续性效果，如 Acolyte of Pain）
    public IReadOnlyList<EventTrigger> EventTriggers { get; init; }

    // 光环（Aura）——持续修改他人属性
    public AuraDefinition? Aura { get; init; }

    // 属性脚本（动态属性，如巨人系列根据场面费用变化）
    public Func<GameEntity, int, int>? CostScript { get; init; }
}
```

### 第三层：Phase / Sequence / Step 引擎（最难也最重要）

这是被 90% 的文章跳过的部分。当 Dr. Boom 的战吼（一个 Phase）开始时，Boom Bot 被召唤，每个都触发 Knife Juggler 向敌方随从投掷飞刀。尽管召唤随从是多个 Phase 的 Sequence，但我们已经在一个 Phase 内部，所以整个战吼保持在同一个 Phase 内，而非开始和停止。

当多个随从同时被消灭时，它们的死亡会按照它们进场的顺序逐一处理。一旦所有被消灭的随从从战场移除，每个死亡会依次处理，包括激活亡语、触发效果以及密语——这些都被视为与亡语完全相同的处理。

```csharp
// ===== Phase / Step 引擎 =====
public class GameEngine
{
    // 关键：死亡检查不是即时的，而是在每个 Phase 结束时批量处理
    private readonly Queue<GameAction> _pendingActions = new();
    private int _phaseDepth = 0;

    public async Task ExecuteAction(GameAction action)
    {
        _phaseDepth++;
        await DispatchAction(action);
        _phaseDepth--;

        // 只有在最外层 Phase 结束时才做死亡检查
        if (_phaseDepth == 0)
            await RunDeathStep();
    }

    private async Task RunDeathStep()
    {
        // 批量收集所有 HP <= 0 的随从
        // 按进场顺序排序，逐一处理亡语
        var deadMinions = _gameState.AllMinions
            .Where(m => m.GetTag(GameTag.HEALTH) <= 0)
            .OrderBy(m => m.GetTag(GameTag.ENTITY_ID)) // 进场顺序
            .ToList();

        if (!deadMinions.Any()) return;

        foreach (var minion in deadMinions)
        {
            // 从战场移除
            minion.SetTag(GameTag.ZONE, (int)Zone.GRAVEYARD);

            // 触发亡语（本身是一个新 Phase，递归调用 ExecuteAction）
            if (minion.HasTag(GameTag.DEATHRATTLE))
            {
                var def = _cardRegistry.Get(minion.CardId);
                foreach (var dAction in def.DeathrattleActions)
                    await ExecuteAction(dAction.WithSource(minion));
            }

            // 触发其他死亡监听（Cult Master、Duplicate 等）
            await _eventBus.Emit(new MinionDiedEvent(minion));
        }

        // 亡语可能产生新的死亡，递归检查
        await RunDeathStep();

        // 最后更新 Aura
        UpdateAllAuras();
    }
}
```

### 第四层：类型化事件总线（含优先级）

像 Brann Bronzebeard、Baron Rivendare 和 Drakkari Enchanter 这类会多次触发效果的卡牌，在 Event 开始前就会检查一次。这意味着如果一个战吼/亡语/回合结束效果召唤了它们，本次触发不会再次生效。但之后的亡语/回合结束效果可以受益，因为光环（Aura）在你召唤它们时已经更新。

这决定了事件总线**必须有优先级**，且每次事件前必须做一次快照：

```csharp
public class PriorityEventBus
{
    // (事件类型, 优先级) -> 订阅者列表
    // 优先级决定触发顺序，默认按进场顺序
    private readonly SortedDictionary<int, List<EventSubscription>> _subs = new();

    public void Subscribe<T>(
        Func<T, Task> handler,
        int priority = 0  // 0 = 默认按进场顺序; 负数 = 更早触发
    ) where T : GameEvent { ... }

    public async Task Emit<T>(T gameEvent) where T : GameEvent
    {
        // 关键：在事件触发前做一次快照
        // 避免"事件处理中途新订阅者被加入"的问题
        var snapshot = _subs.Values
            .SelectMany(x => x)
            .Where(s => s.EventType == typeof(T))
            .OrderBy(s => s.Priority)
            .ThenBy(s => s.SourceEntityId) // 相同优先级按进场顺序
            .ToList();

        foreach (var sub in snapshot)
        {
            // 检查订阅者的 source entity 是否仍然存活
            if (sub.SourceEntity is not null &&
                sub.SourceEntity.GetTag(GameTag.ZONE) != (int)Zone.PLAY)
                continue;

            await ((Func<T, Task>)sub.Handler)(gameEvent);
        }
    }
}

// 事件类型（最小完整集）
public abstract record GameEvent;
public record MinionSummonedEvent(GameEntity Minion) : GameEvent;
public record MinionDiedEvent(GameEntity Minion) : GameEvent;
public record DamageDealtEvent(GameEntity Source, GameEntity Target, int Amount) : GameEvent;
public record HealEvent(GameEntity Target, int Amount) : GameEvent;
public record CardPlayedEvent(GameEntity Player, GameEntity Card) : GameEvent;
public record SpellCastEvent(GameEntity Spell, GameEntity? Target) : GameEvent;
public record TurnStartEvent(GameEntity Player) : GameEvent;
public record TurnEndEvent(GameEntity Player) : GameEvent;
public record AttackEvent(GameEntity Attacker, GameEntity Defender) : GameEvent;
public record SecretCheckEvent(GameEntity TriggerSource) : GameEvent;
```

### 与文章框架的对比总结

|维度|知乎文章的 ActionSystem|生产级架构|
|---|---|---|
|卡牌效果描述|`switch` 硬编码|Selector DSL + Atomic Actions 组合|
|死亡检查时机|即时触发|Phase 结束后批量处理|
|事件顺序|注册顺序|优先级 + 进场顺序快照|
|取消订阅|有 Bug（永远失败）|Priority Queue，实例级生命周期绑定|
|全局状态|`static` 字典污染|局内实例，随 GameSession 销毁|
|网络同步|无|TAG_CHANGE delta，服务端权威|
|可测试性|依赖 MonoBehaviour|纯逻辑层，无引擎耦合|
|Aura 更新|无|每个最外层 Phase 结束后重算|

正如 Fireplace 作者所说，逆向工程炉石的过程是他编程生涯中最有启发性的体验——炉石是一个设计极为严谨的游戏，规则极具逻辑性，很少有 hack。 这套严谨的规则体系正是商业级架构应该对齐的标准。
