代码例子：满足玩家按键，能够触发角色动作，并实现能方便自定义按键映射，以及可以撤回操作
1. 基础的 `Command` 类：
```cpp
class Command
{
public:
	virtual ~Command();
	virtual void excute() = 0;
	virtual void undo() = 0;
}
```

2. 输入处理程序代码，每一步会生成一个 `Command` 实例
```cpp
Command* handleInput()
{
　Unit* unit = getSelectedUnit();

　if (isPressed(BUTTON_UP)) {
　　// Move the unit up one.
　　int destY = unit->y() - 1;
　　return new MoveUnitCommand(
　　　　unit, unit->x(), destY);
　}

　if (isPressed(BUTTON_DOWN)) {
　　// Move the unit down one.
　　int destY = unit->y() + 1;
　　return new MoveUnitCommand(
　　　　unit, unit->x(), destY);
　}

　// Other moves...

　return NULL;
}
```

3. `MoveUnitCommand`
```cpp
class MoveUnitCommand : public Command
{
public:
	MoveUnityCommand(Unit* unit, int x, int y)
	: unit_(unit), x_(x), y_(y), xBefore_(0), yBefore_(0)
	{}
	
	virtual void excute()
	{
		xBefore_ = unit_->x();
		yBefore_ = unit_->y();
		unit_->moveTo(x_, y_);
	}
	
	virtual void undo()
	{
		unit->moveTo(xBefore_, yBefore_)
	}
	
private:
	Unit* unit_;
	int x_;
	int y_;
	int xBefore_;
	int yBefore_;
}
```

这样之后就可以得到 Command Stack，只要遍历 `Command->undo()`就可以实现多次撤销
